function fetchSnapshot() {
    try {
        if (typeof getvar === 'function') {
            for (var i = -1; i >= -6; i--) {
                var v = getvar('stat_data.渐变带', { scope: 'message', withMsg: { id: i } });
                if (v && v['主角']) return v;
            }
        }
    } catch (e) {}

    if (typeof variables !== 'undefined' && variables) {
        if (variables.stat_data && variables.stat_data['渐变带'] && variables.stat_data['渐变带']['主角']) {
            return variables.stat_data['渐变带'];
        }
        if (variables['渐变带'] && variables['渐变带']['主角']) {
            return variables['渐变带'];
        }
    }
    return null;
}

    [10000000, '≈2.4公斤TNT爆炸'],
    [100000000, '≈24公斤TNT爆炸'],
    [1000000000, '≈半吨TNT爆炸'],
    [1e12, '≈战术核弹破坏'],
    [Infinity, '≈战略级天灾']
];

function getAnchor(kJ) {
    var J = (kJ || 0) * 1000;
    for (var i = 0; i < ANCHOR_TABLE.length; i++) {
        if (J <= ANCHOR_TABLE[i][0]) return ANCHOR_TABLE[i][1];
    }
    return '微弱感应';
}
